rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn(){ return request.auth != null; }
    function userDoc(){ return get(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function activeUser(){ return signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && userDoc().data.active == true; }
    function isAdmin(){ return activeUser() && userDoc().data.role == 'admin'; }
    function isTeacher(){ return activeUser() && userDoc().data.role == 'guru'; }
    function isStudent(){ return activeUser() && userDoc().data.role == 'siswa'; }
    function isStaff(){ return isAdmin() || isTeacher(); }
    function isHomeroom(){ return isTeacher() && userDoc().data.isHomeroom == true; }
    function isAnonymous(){ return signedIn() && request.auth.token.firebase.sign_in_provider == 'anonymous'; }
    function chatGroupAllowed(groupId){
      return isStaff() && (
        groupId == 'announcements' || groupId == 'all-teachers'
        || (groupId == 'homerooms' && (isAdmin() || isHomeroom()))
        || (groupId == 'grade-7' && (isAdmin() || (isHomeroom() && userDoc().data.homeroomClass.matches('^7.*'))))
        || (groupId == 'grade-8' && (isAdmin() || (isHomeroom() && userDoc().data.homeroomClass.matches('^8.*'))))
        || (groupId == 'grade-9' && (isAdmin() || (isHomeroom() && userDoc().data.homeroomClass.matches('^9.*'))))
      );
    }

    match /studentAccess/{nis} {
      allow get: if isAdmin() || isAnonymous();
      allow list: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if isAdmin()
        || (isAnonymous() && request.auth.uid == uid
          && request.resource.data.role == 'siswa'
          && request.resource.data.active == true
          && request.resource.data.approved == true
          && request.resource.data.loginId is string
          && request.resource.data.studentId is string
          && request.resource.data.classId is string)
        || (signedIn() && request.auth.uid == uid
          && request.resource.data.keys().hasOnly(['loginId','name','role','approved','active','rejected','requestedType','requestedHomeroomClass','isHomeroom','homeroomClass','createdAt'])
          && request.resource.data.loginId is string && request.resource.data.name is string
          && request.resource.data.role == 'guru' && request.resource.data.approved == false
          && request.resource.data.active == false && request.resource.data.rejected == false
          && request.resource.data.requestedType in ['guru','wali']
          && request.resource.data.isHomeroom == false && request.resource.data.homeroomClass == '');
      allow update: if isAdmin();
      allow delete: if isAdmin() || (isStudent() && request.auth.uid == uid);
    }

    match /classes/{id}{ allow read: if isStaff(); allow write: if isAdmin(); }
    match /students/{id}{
      allow read: if isStaff() || (isStudent() && userDoc().data.studentId == id) || isAnonymous();
      allow create, update, delete: if isAdmin();
    }
    match /records/{id}{ allow read: if isStaff(); allow create, update: if isStaff(); allow delete: if isAdmin(); }
    match /cleanliness/{id}{ allow read: if isStaff(); allow create, update: if isStaff(); allow delete: if isAdmin(); }
    match /settings/{id}{ allow read: if isStaff(); allow write: if isAdmin(); }
    match /analysisPeriods/{id}{ allow read: if isAdmin() || (isHomeroom() && resource.data.sharedToHomeroom == true); allow write: if isAdmin(); }
    match /analysisPeriodRevisions/{id}{ allow read, create: if isAdmin(); allow update, delete: if false; }
    match /auditLogs/{id}{ allow read, create: if isAdmin(); allow update, delete: if false; }
    match /recordResetArchive/{id}{ allow read, create: if isAdmin(); allow update, delete: if false; }
    match /cleanlinessResetArchive/{id}{ allow read, create: if isAdmin(); allow update, delete: if false; }

    match /chatMessages/{id}{
      allow read: if chatGroupAllowed(resource.data.groupId);
      allow create: if isStaff() && chatGroupAllowed(request.resource.data.groupId)
        && ((request.resource.data.type == 'bot' && isAdmin() && request.resource.data.senderUid == 'ecobot')
        || (request.resource.data.type != 'bot' && request.resource.data.senderUid == request.auth.uid
        && (request.resource.data.groupId != 'announcements' || isAdmin())));
      allow update: if isAdmin() || (isStaff() && resource.data.senderUid == request.auth.uid);
      allow delete: if isAdmin() || (isStaff() && resource.data.senderUid == request.auth.uid);
    }
    match /chatReactions/{id}{
      allow read: if chatGroupAllowed(resource.data.groupId);
      allow create: if isStaff() && request.resource.data.uid == request.auth.uid && chatGroupAllowed(request.resource.data.groupId);
      allow update: if false;
      allow delete: if isAdmin() || (isStaff() && resource.data.uid == request.auth.uid);
    }
    match /chatReads/{id}{
      allow read: if isStaff() && (resource.data.uid == request.auth.uid || isAdmin());
      allow create, update: if isStaff() && request.resource.data.uid == request.auth.uid && chatGroupAllowed(request.resource.data.groupId);
      allow delete: if isAdmin() || (isStaff() && resource.data.uid == request.auth.uid);
    }

    match /assignments/{taskId}{
      // Siswa wajib membaca dengan query targetClasses array-contains classId.
      allow read: if isStaff() || (isStudent()
        && resource.data.targetClasses is list
        && userDoc().data.classId in resource.data.targetClasses);
      allow create, update, delete: if isAdmin();
    }
    match /taskSubmissions/{id}{
      allow read: if isAdmin()
        || (isHomeroom() && resource.data.classId == userDoc().data.homeroomClass)
        || (isStudent()
          && (resource.data.nis == userDoc().data.loginId
            || (resource.data.memberNis is list && userDoc().data.loginId in resource.data.memberNis)));

      allow create: if isStudent()
        && request.resource.data.studentUid == request.auth.uid
        && request.resource.data.nis == userDoc().data.loginId
        && request.resource.data.classId == userDoc().data.classId
        && (request.resource.data.taskType == 'individual'
          || (request.resource.data.taskType == 'group'
            && request.resource.data.memberNis is list
            && userDoc().data.loginId in request.resource.data.memberNis));

      allow update: if (isStudent()
        && resource.data.classId == userDoc().data.classId
        && (resource.data.nis == userDoc().data.loginId
          || (resource.data.memberNis is list && userDoc().data.loginId in resource.data.memberNis))
        && request.resource.data.taskId == resource.data.taskId
        && request.resource.data.classId == resource.data.classId
        && request.resource.data.taskType == resource.data.taskType
        && request.resource.data.groupNo == resource.data.groupNo
        && request.resource.data.groupKey == resource.data.groupKey
        && request.resource.data.memberNis == resource.data.memberNis
        && request.resource.data.memberNames == resource.data.memberNames
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'answers','reflection','status','submittedAt','submittedByNis','submittedByName',
          'updatedAt','updatedByNis','studentUid','studentId','nis','studentName'
        ]))
        || ((isAdmin() || (isHomeroom() && resource.data.classId == userDoc().data.homeroomClass))
        && request.resource.data.taskId == resource.data.taskId
        && request.resource.data.classId == resource.data.classId
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'score','feedback','status','gradedAt','gradedByUid','gradedByName','updatedAt'
        ]));
      allow delete: if isAdmin();
    }

    match /{document=**}{ allow read, write: if false; }
  }
}
